    // Debounce
    function debounce(func, delay) {
        let timer;
        return function(...args) {
            clearTimeout(timer);
            timer = setTimeout(() => func.apply(this, args), delay);
        };
    }

    // Arama vurgulama (Highlight)
    function highlightText(text, term) {
        if (!term) return text;
        const regex = new RegExp(`(${term})`, 'gi');
        return text.replace(regex, '<mark>$1</mark>');
    }
    
    // Okuma süresi hesaplama
    function getReadingTime(text) {
        const words = text.trim().split(/\s+/).length;
        const minutes = Math.ceil(words / 200); // Ortalama 200 kelime/dk
        return minutes;
    }

    function entegreBloggerYorumlari() {
        const bloggerComments = document.getElementById('comments') ||

        if (bloggerComments && bizimKutu) {

            bizimKutu.classList.add('visible');
        }

    async function loadIrfanData() {
        qaGrid.innerHTML = '<div class="no-results"><i class="fas fa-spinner fa-spin"></i> Kadim sırlar ve hakikatler yükleniyor...</div>';
        
        try {
            const cachedData = sessionStorage.getItem('tabirly_qa_cache');
            if (cachedData) {
                qaData = JSON.parse(cachedData);
            } else {
                const response = await fetch(DATA_URL);
                if (!response.ok) throw new Error('Ağ yanıtı sorunlu');
                qaData = await response.json();
                sessionStorage.setItem('tabirly_qa_cache', JSON.stringify(qaData));
            }
        } catch (error) {
            console.warn('Veriler yüklenemedi.', error);
            qaGrid.innerHTML = '<div class="no-results">Enerji akışında bir kesinti oldu. Lütfen sayfayı yenileyin.</div>';
            return;
        }

        generateFilterButtons(qaData);
        filterData();
        generateFAQSchema();
        setTimeout(entegreBloggerYorumlari, 500); 

        // URL'de doğrudan soru ID'si varsa (paylaşılan bir linkten gelindiyse) otomatik aç
        const urlParams = new URLSearchParams(window.location.search);
        const directQuestionId = urlParams.get('soru');
        if (directQuestionId) {
            const idNum = parseInt(directQuestionId, 10);
            if (qaData.some(q => q.id === idNum)) {
                setTimeout(() => { openModal(idNum); }, 400); // Sistem yüklendikten hemen sonra modalı aç
            }
        }
    }

    function generateFilterButtons(data) {
        filterContainer.innerHTML = '';
        const totalCount = data.length;

        const allBtn = document.createElement('button');
        allBtn.className = 'filter-btn active';
        allBtn.setAttribute('data-filter', 'all');
        allBtn.setAttribute('aria-pressed', 'true');
        allBtn.innerHTML = `Tüm Sırlar <span class="count-badge">${totalCount}</span>`;
        filterContainer.appendChild(allBtn);

        const categoryCounts = {};
        data.forEach(item => {
            categoryCounts[item.category] = (categoryCounts[item.category] || 0) + 1;
        });

        Object.keys(categoryCounts).forEach(category => {
            const btn = document.createElement('button');
            btn.className = 'filter-btn';
            btn.setAttribute('data-filter', category);
            btn.setAttribute('aria-pressed', 'false');
            btn.innerHTML = `${category} <span class="count-badge">${categoryCounts[category]}</span>`;
            filterContainer.appendChild(btn);
        });

        filterContainer.addEventListener('click', (e) => {
            const targetBtn = e.target.closest('.filter-btn');
            if (targetBtn) {
                document.querySelectorAll('.filter-btn').forEach(b => {
                    b.classList.remove('active');
                    b.setAttribute('aria-pressed', 'false');
                });
                targetBtn.classList.add('active');
                targetBtn.setAttribute('aria-pressed', 'true');
                currentFilter = targetBtn.getAttribute('data-filter');
                filterData();
            }
        });
    }

    function createCardElement(item) {
        const card = document.createElement('div');
        card.className = 'qa-card';
        card.setAttribute('data-id', item.id);
        card.setAttribute('role', 'listitem');
        
        if (readItems.includes(item.id)) {
            card.classList.add('is-read');
        }
        
        const maxLength = 250;
        let displayAnswer = item.answer.replace(/\n/g, '<br>');
        let readMoreHTML = '';
        
        if (item.answer.length > maxLength) {
            displayAnswer = item.answer.substring(0, maxLength).replace(/\n/g, '<br>') + '...';
            readMoreHTML = `<br><button class="read-more-btn" onclick="openModal(${item.id}, event)" aria-label="Cevabın tamamını oku">Tamamını Oku</button>`;
        }

        const userFeedback = feedbackItems[item.id];
        const isLikeActive = userFeedback === 'like' ? 'like-active' : '';
        const isDislikeActive = userFeedback === 'dislike' ? 'dislike-active' : '';
        
        const highlightedTitle = highlightText(item.question, searchTerm);
        const readingTime = getReadingTime(item.answer);

        card.innerHTML = `

                    </div>

                <h3 class="card-question">${highlightedTitle}</h3>
                <div class="card-indicator"><i class="fas fa-chevron-down"></i></div>

            <div class="card-answer-wrapper" aria-hidden="true">
                <div class="card-answer-content">

    function filterData() {
        filteredData = qaData;

        if (currentFilter !== 'all') {
            filteredData = filteredData.filter(item => item.category === currentFilter);
        }

        if (searchTerm) {
            const lowerTerm = searchTerm.toLowerCase();
            filteredData = filteredData.filter(item => 
                item.question.toLowerCase().includes(lowerTerm) || 
                item.answer.toLowerCase().includes(lowerTerm)
            );
        }
        
        currentIndex = 0;
        qaGrid.innerHTML = '';
        
        if(filteredData.length === 0) {
            qaGrid.innerHTML = '<div class="no-results">Bu frekansta bir yanıt bulunamadı. Lütfen aramanızı farklı kelimelerle tekrarlayın.</div>';
            scrollSentinel.style.display = 'none';
            return;
        }
        
        scrollSentinel.style.display = 'block';
        loadMoreCards();
    }
    
    function loadMoreCards() {
        if (currentIndex >= filteredData.length) return;
        
        const fragment = document.createDocumentFragment();
        const chunk = filteredData.slice(currentIndex, currentIndex + ITEMS_PER_LOAD);
        
        chunk.forEach(item => {
            fragment.appendChild(createCardElement(item));
        });
        
        qaGrid.appendChild(fragment);
        currentIndex += ITEMS_PER_LOAD;
        
        if(currentIndex >= filteredData.length) {
            scrollSentinel.style.display = 'none';
        }
    }

    // Sonsuz kaydırma için IntersectionObserver
    const observer = new IntersectionObserver((entries) => {
        if (entries[0].isIntersecting) {
            loadMoreCards();
        }
    }, { rootMargin: '200px' });
    observer.observe(scrollSentinel);

    function toggleCard(card, id) {

            activeCard.querySelector('.card-answer-wrapper').setAttribute('aria-hidden', 'true');
        });

        if (!isActive) {
            card.classList.add('active');
            header.setAttribute('aria-expanded', 'true');
            wrapper.setAttribute('aria-hidden', 'false');
        }
        
        markAsRead(id, card);
    }
    
    // Sesli Okuma (TTS) Motoru
    window.toggleTTS = function(text, event, btnId) {
        if(event) event.stopPropagation();
        const btn = document.getElementById(btnId);
        
        if (window.speechSynthesis.speaking) {
            window.speechSynthesis.cancel();
            document.querySelectorAll('.tts-btn').forEach(b => {
                b.innerHTML = '<i class="fas fa-volume-up"></i> Dinle';
                b.dataset.playing = "false";
            });
            if (btn && btn.dataset.playing === "true") {
                return; // Eğer çalışan butona tıklandıysa sadece durdur.
            }
        }
        
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = 'tr-TR';
        utterance.rate = 0.95; 
        
        utterance.onend = () => {
            if(btn) {
                btn.innerHTML = '<i class="fas fa-volume-up"></i> Dinle';
                btn.dataset.playing = "false";
            }
        };
        
        if(btn) {
            btn.innerHTML = '<i class="fas fa-stop"></i> Durdur';
            btn.dataset.playing = "true";
        }
        
        window.speechSynthesis.speak(utterance);
    };

    // Web Share API
    window.shareQuestion = async function(id, event) {
        event.stopPropagation();
        const item = qaData.find(q => q.id === id);
        if (!item) return;

        let pageUrl = window.location.href;
        // Mevcut URL'deki eski parametreleri temizle (üst üste binmemesi için)
        pageUrl = pageUrl.split('?')[0]; 

        const canonicalMeta = document.querySelector('link[rel="canonical"]');
        if (canonicalMeta && canonicalMeta.href) {
            pageUrl = canonicalMeta.href;
        }
        
        // ÖNİZLEME KORUMASI VE GARANTİ LİNK:
        // Eğer sistem 'tabirly.com'da olmadığını fark ederse (örneğin şu anki test ekranında)
        // veya link okumada sorun yaşarsa aşağıdaki linki kullanır.
        // LÜTFEN AŞAĞIDAKİ LİNKİ BU ARACI YAYINLAYACAĞIN SAYFANIN TAM LİNKİ İLE DEĞİŞTİR.
        if (!pageUrl.includes('tabirly.com')) {
             pageUrl = 'https://tr.tabirly.com/p/islami-ve-ezoterik-soru-cevap.html'; 
        }

        // Linke doğrudan soruyu açacak parametreyi ekliyoruz (?soru=ID)
        const urlObj = new URL(pageUrl);
        urlObj.searchParams.set('soru', item.id);
        const finalUrl = urlObj.toString();

        const shareData = {
            title: 'Tabirly - İslami ve Ezoterik Meraklar',
            text: `Soru: ${item.question}\n\nDetaylı cevabı okumak için Tabirly'yi ziyaret et:`,
            url: finalUrl
        };

        if (navigator.share) {
            try {
                await navigator.share(shareData);
            } catch (err) {
                console.log('Paylaşım iptal edildi veya desteklenmiyor.', err);
            }
        } else {
            // Desteklemeyen tarayıcılar için zarif kopyalama işlemi
            const fallbackInput = document.createElement('textarea');
            fallbackInput.value = `${shareData.text} \n${shareData.url}`;
            document.body.appendChild(fallbackInput);
            fallbackInput.select();
            document.execCommand('copy');
            document.body.removeChild(fallbackInput);
            
            showToast('Bağlantı panoya kopyalandı! ✦');
        }
    };

    // Şık Bildirim Mesajı (Toast Notification)
    function showToast(message) {
        let toast = document.getElementById('irfan-toast');
        if (!toast) {
            toast = document.createElement('div');
            toast.id = 'irfan-toast';
            toast.style.cssText = 'position:fixed; bottom:30px; left:50%; transform:translateX(-50%); background:var(--text-dark-sage); color:var(--accent-gold); padding:12px 24px; border-radius:8px; z-index:9999999; font-family:"Outfit", sans-serif; font-weight:600; box-shadow:0 10px 30px rgba(0,0,0,0.2); transition:opacity 0.4s ease; opacity:0; pointer-events:none;';
            document.body.appendChild(toast);
        }
        toast.textContent = message;
        toast.style.display = 'block';
        setTimeout(() => toast.style.opacity = '1', 10);
        setTimeout(() => { 
            toast.style.opacity = '0'; 
            setTimeout(() => toast.style.display = 'none', 400); 
        }, 3000);
    }

    window.sendFeedback = async function(id, type, event) {
        event.stopPropagation();
        
        const container = document.getElementById(`feedback-${id}`);
        const buttons = container.querySelectorAll('.feedback-btn');
        const thanksMsg = container.querySelector('.feedback-thanks');

        if (feedbackItems[id] === type) {
            showThanks(thanksMsg);
            return;
        }

        buttons.forEach(btn => {
            btn.classList.remove('like-active', 'dislike-active');
            if (btn.textContent.includes('Beğendim') && type === 'like') {
                btn.classList.add('like-active');
            } else if (btn.textContent.includes('Beğenmedim') && type === 'dislike') {
                btn.classList.add('dislike-active');
            }
        });
        
        showThanks(thanksMsg);

        feedbackItems[id] = type;
        localStorage.setItem('tabirly_feedbacks', JSON.stringify(feedbackItems));

        const questionItem = qaData.find(q => q.id === id);
        const questionTitle = questionItem ? questionItem.question : "Bilinmeyen Soru";

        try {
            const formData = new URLSearchParams();
            formData.append(ENTRY_FEEDBACK, type === 'like' ? 'Beğendim' : 'Beğenmedim');
            formData.append(ENTRY_PAGE, window.location.href);
            formData.append(ENTRY_TITLE, `ID: ${id} - ${questionTitle}`);
            formData.append(ENTRY_UA, navigator.userAgent);

            fetch(GOOGLE_FORM_URL, {
                method: 'POST',
                mode: 'no-cors',
                headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
                body: formData
            });
        } catch (error) {}
    };

    function showThanks(el) {
        el.classList.remove('hidden-el');
        setTimeout(() => el.classList.add('hidden-el'), 2000);
    }

    function markAsRead(id, cardElement) {
        if (!readItems.includes(id)) {
            readItems.push(id);
            localStorage.setItem('tabirly_read_qas', JSON.stringify(readItems));
            if(cardElement) cardElement.classList.add('is-read');
        }
    }

    window.openModal = function(id, event) {
        if(event) event.stopPropagation(); 
        
        const item = qaData.find(q => q.id === id);
        if(item) {
            const cardEl = document.querySelector(`.qa-card[data-id="${id}"]`);
            markAsRead(id, cardEl);

            modalCategory.textContent = item.category;
            modalQuestion.textContent = item.question;
            modalAnswer.innerHTML = item.answer.replace(/\n/g, '<br>');
            
            // İlgili Soruları (Benzer Sırlar) Bul ve Ekle
            modalRelatedQuestions.innerHTML = '';
            const related = qaData.filter(q => q.category === item.category && q.id !== id)
                                  .sort(() => 0.5 - Math.random())
                                  .slice(0, 2);
                                  
            if (related.length > 0) {
                const title = document.createElement('h4');
                title.className = 'related-questions-title';
                title.textContent = 'Bunu okuyanlar bu sırlara da göz attı:';
                modalRelatedQuestions.appendChild(title);
                
                related.forEach(r => {
                    const link = document.createElement('div');
                    link.className = 'related-question-link';
                    link.textContent = r.question;
                    link.onclick = () => {
                        window.speechSynthesis.cancel(); // Yeni soruya geçerken sesi kapat
                        openModal(r.id);
                    };
                    modalRelatedQuestions.appendChild(link);
                });
            }
            
            irfanModal.classList.add('active');
            irfanModal.setAttribute('aria-hidden', 'false');
            document.body.classList.add('modal-open');
        }
    };

    function closeModal() {
        irfanModal.classList.remove('active');
        irfanModal.setAttribute('aria-hidden', 'true');
        document.body.classList.remove('modal-open');
        window.speechSynthesis.cancel(); // Modal kapanırken sesi iptal et
        if (modalTtsBtn) {
            modalTtsBtn.innerHTML = '<i class="fas fa-volume-up"></i> Dinle';
            modalTtsBtn.dataset.playing = "false";
        }
    }

    modalCloseBtn.addEventListener('click', closeModal);
    irfanModal.addEventListener('click', (e) => {
        if (e.target === irfanModal) closeModal();
    });
    document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape' && irfanModal.classList.contains('active')) closeModal();
    });

    function generateFAQSchema() {
        const oldSchema = document.querySelector('script[type="application/ld+json"]');
        if(oldSchema) oldSchema.remove();

        const schema = {
            "@context": "https://schema.org",
            "@type": "FAQPage",
            "mainEntity": qaData.map(item => ({
                "@type": "Question",
                "name": item.question,
                "acceptedAnswer": {
                    "@type": "Answer",
                    "text": item.answer
                }
            }))
        };

        const script = document.createElement('script');
        script.type = 'application/ld+json';
        script.text = JSON.stringify(schema);
        document.head.appendChild(script);
    }

    document.addEventListener('DOMContentLoaded', loadIrfanData);
